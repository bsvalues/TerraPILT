<script>
  // Initialize the spreadsheet
  const container = document.getElementById('pilt-grid');
  const hot = new Handsontable(container, {
    data: piltResults,
    rowHeaders: true,
    colHeaders: ['District', 'Assessed Value', 'Levy Rate', 'Base PILT', 'Deduction', 'PILT Due'],
    columns: [
      { data: 'district' },
      { 
        data: 'assessedValue',
        type: 'numeric',
        numericFormat: { pattern: '$0,0.00', culture: 'en-US' }
      },
      { 
        data: 'levyRate',
        type: 'numeric',
        numericFormat: { pattern: '0.0000' }
      },
      { 
        data: 'basePILT',
        type: 'numeric',
        numericFormat: { pattern: '$0,0.00', culture: 'en-US' },
        readOnly: true 
      },
      { 
        data: 'deduction',
        type: 'numeric',
        numericFormat: { pattern: '$0,0.00', culture: 'en-US' }
      },
      { 
        data: 'piltDue',
        type: 'numeric',
        numericFormat: { pattern: '$0,0.00', culture: 'en-US' },
        readOnly: true
      }
    ],
    contextMenu: true,
    licenseKey: 'non-commercial-and-evaluation',
    afterChange: function(changes, source) {
      if (source !== 'loadData') {
        // Recalculate PILT values after user edits
        recalculatePILT();
      }
    }
  });
  
  // Function to recalculate PILT values
  function recalculatePILT() {
    const data = hot.getData();
    const districtData = data.map((row, index) => ({
      district: row[0],
      assessedValue: parseFloat(row[1]) || 0,
      levyRate: parseFloat(row[2]) || 0,
      deduction: parseFloat(row[4]) || 0
    }));
    
    // Run calculations
    const piltResults = window.calculatePILT(districtData);
    
    // Update calculated columns
    piltResults.forEach((result, index) => {
      hot.setDataAtCell(index, 3, result.basePILT, 'calculation');
      hot.setDataAtCell(index, 5, result.piltDue, 'calculation');
    });
    
    // Update summary and charts
    updateSummary(piltResults);
    updateCharts(piltResults);
  }
</script>
