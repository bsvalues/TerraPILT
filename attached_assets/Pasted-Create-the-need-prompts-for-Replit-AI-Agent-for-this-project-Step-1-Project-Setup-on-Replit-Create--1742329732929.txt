# calculations.py
import pandas as pd

def calculate_pilt(df: pd.DataFrame, deductions: dict = None) -> pd.DataFrame:
    """
    Calculate the PILT due for each district.
    
    :param df: DataFrame with columns: 'District', 'Assessed_Value', 'Levy_Rate'
    :param deductions: Optional dictionary with district as key and deduction amount as value.
    :return: DataFrame with additional 'PILT_Due' column.
    """
    df['Base_PILT'] = (df['Assessed_Value'] / 1000) * df['Levy_Rate']
    if deductions:
        df['Deduction'] = df['District'].map(deductions).fillna(0)
    else:
        df['Deduction'] = 0
    df['PILT_Due'] = df['Base_PILT'] - df['Deduction']
    return df

st.title("Benton County PILT Dashboard")

# Option to load data either from SQL or an uploaded Excel file
data_source = st.sidebar.radio("Select Data Source", ("SQL Database", "Excel File"))

if data_source == "Excel File":
    file = st.sidebar.file_uploader("Upload PILT Workbook (Excel)", type=["xlsx"])
    if file:
        df = pd.read_excel(file, sheet_name="2023")
else:
    # Define your SQL query here (you can use one of the refactored queries from before)
    sql_query = """
    WITH CurrentYear AS (
        SELECT appr_yr AS current_year, appr_yr - 1 AS prev_year
        FROM pacs_oltp.dbo.pacs_system
    ),
    NonExemptProps AS (
        SELECT prop_id
        FROM property_exemption
        CROSS JOIN CurrentYear
        WHERE exmpt_tax_yr = prev_year
          AND exmpt_type_cd IN ('SNR/DSBL', 'EX')
    )
    SELECT DISTINCT
        pv.prop_id,
        p.geo_id,
        pv.assessed_val AS Assessed_Value,
        pv.market,
        'Current Expense' AS District,
        ld.size_acres,
        ld.mkt_val_source,
        ld.land_seg_mkt_val,
        ld.mkt_flat_val,
        ld.mkt_adj_val,
        ld.ag_unit_price,
        ld.ag_val,
        ld.ag_adj_val,
        ld.ag_calc_val,
        ld.ag_flat_val,
        ld.ag_val_source,
        0.9007089219 AS Levy_Rate  -- Example rate for Current Expense
    FROM property_val AS pv WITH (NOLOCK)
    INNER JOIN prop_supp_assoc AS psa WITH (NOLOCK)
        ON pv.prop_id = psa.prop_id
       AND pv.prop_val_yr = psa.owner_tax_yr
       AND pv.sup_num = psa.sup_num
    INNER JOIN property AS p WITH (NOLOCK)
        ON pv.prop_id = p.prop_id
    INNER JOIN land_detail AS ld WITH (NOLOCK)
        ON pv.prop_id = ld.prop_id
       AND pv.prop_val_yr = ld.prop_val_yr
       AND pv.sup_num = ld.sup_num
       AND ld.sale_id = 0
       AND ld.land_type_cd IN ('4', '41', '42', '45', '46', '47')
    CROSS JOIN (SELECT appr_yr AS current_year FROM pacs_oltp.dbo.pacs_system) CY
    WHERE pv.prop_val_yr = CY.current_year
      AND pv.prop_inactive_dt IS NULL
      AND pv.prop_id NOT IN (SELECT prop_id FROM NonExemptProps)
    ORDER BY p.geo_id;
    """
    try:
        df = load_data(sql_query)
    except Exception as e:
        st.error(f"Error loading data from SQL: {e}")
        df = pd.DataFrame()

if not df.empty:
    st.write("### Raw Data", df)
    
    # What-if sliders for adjusting levy rates
    st.sidebar.header("Adjust Levy Rates")
    default_rate = df['Levy_Rate'].iloc[0] if 'Levy_Rate' in df.columns else 0.9
    new_rate = st.sidebar.slider("Levy Rate (Current Expense)", 0.0, 2.0, default_rate, 0.01)
    
    # Update DataFrame with new rate for demonstration (you can expand this for each district)
    df.loc[df['District'] == 'Current Expense', 'Levy_Rate'] = new_rate
    
    # Calculate PILT using enhanced calculation logic
    deductions = {}  # Populate with any specific deductions if needed
    df = calculate_pilt(df, deductions)
    
    st.write("### Calculated PILT by District", df[['prop_id', 'District', 'PILT_Due']])
    total_pilt = df['PILT_Due'].sum()
    st.write(f"## Total PILT Due: ${total_pilt:,.2f}")
    
    # Year-over-year trend analysis (dummy data for illustration)
    hist_data = pd.DataFrame({
        'Year': [2020, 2021, 2022, 2023],
        'PILT_Due': [3500000, 3600000, 3534858.69, total_pilt]

    fig = px.line(hist_data, x='Year', y='PILT_Due', title='Year-over-Year PILT Trend')
